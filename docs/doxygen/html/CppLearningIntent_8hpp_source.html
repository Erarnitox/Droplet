53[[nodiscard]] inline std::string lowercase_ascii(std::string_view text) {
54 std::string out(text.begin(), text.end());
55 std::transform(
56 out.begin(), out.end(), out.begin(), [](unsigned char c) { return static_cast<char>(std::tolower(c)); });
57 return out;
58}

60[[nodiscard]] inline bool mentions_cpp_learning_intent(std::string_view original_text) {
61 const std::string lower = lowercase_ascii(original_text);
62 for (const std::string_view phrase : k_phrases) {
63 if (lower.find(phrase) != std::string::npos) {
64 return true;
65 }
66 }
67 return false;
68}
